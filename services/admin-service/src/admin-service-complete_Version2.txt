import { APIGatewayProxyHandlerV2 } from 'aws-lambda';
import { DynamoDBClient, UpdateItemCommand, QueryCommand } from '@aws-sdk/client-dynamodb';
import { CloudWatchClient, GetMetricStatisticsCommand } from '@aws-sdk/client-cloudwatch';

const ddb = new DynamoDBClient({});
const cloudwatch = new CloudWatchClient({});

// ============================================================================
// ADMIN AUTH MIDDLEWARE
// ============================================================================

function validateAdminAccess(event: any): boolean {
  const role = event.requestContext.authorizer?.claims?.['cognito:groups'];
  return role?. includes('admin');
}

// ============================================================================
// SYSTEM HEALTH CHECK
// ============================================================================

export const getSystemHealth: APIGatewayProxyHandlerV2 = async (event) => {
  if (!validateAdminAccess(event)) {
    return { statusCode: 403, body: JSON.stringify({ error: 'Access denied' }) };
  }

  try {
    const health = {
      status: 'HEALTHY',
      timestamp: new Date().toISOString(),
      services: {
        api: { status: 'UP' },
        database: { status: 'UP' },
        cache: { status: 'UP' },
        notifications: { status: 'UP' },
      },
      metrics: {
        activeUsers: 0,
        shipmentsInTransit: 0,
        systemLoad: 0,
      },
    };

    // Get CloudWatch metrics
    const cpuMetric = await cloudwatch.send(
      new GetMetricStatisticsCommand({
              Namespace: 'AWS/Lambda',
        MetricName: 'Duration',
        Dimensions: [],
        StartTime: new Date(Date.now() - 3600000),
        EndTime: new Date(),
        Period: 300,
        Statistics: ['Average'],
      }),
    );

    health.metrics.systemLoad = cpuMetric. Datapoints? .[0]?.Average || 0;

    return {
      statusCode: 200,
      body: JSON.stringify(health),
    };
  } catch (error) {
    console.error('Health check error:', error);
    return {
      statusCode: 500,
      body: JSON.stringify({ status: 'UNHEALTHY' }),
    };
  }
};

// ============================================================================
// MERCHANT MANAGEMENT
// ============================================================================

export const suspendMerchant: APIGatewayProxyHandlerV2 = async (event) => {
  if (!validateAdminAccess(event)) {
    return { statusCode: 403, body: JSON. stringify({ error: 'Access denied' }) };
  }

  try {
    const merchantId = event.pathParameters?.merchantId;
    const reason = (JSON.parse(event.body || '{}')).reason;

    await ddb.send(
      new UpdateItemCommand({
        TableName: 'Merchants',
        Key: { PK: { S: merchantId!  }, SK: { S: 'METADATA' } },
        UpdateExpression: 'SET #status = :status, suspensionReason = :reason, suspendedAt = :now',
        ExpressionAttributeNames: { '#status': 'status' },
        ExpressionAttributeValues: {
          ':status': { S: 'SUSPENDED' },
          ':reason': { S: reason },
          ':now': { S: new Date().toISOString() },
        },
      }),
    );

    return {
      statusCode: 200,
      body: JSON.stringify({ message: `Merchant ${merchantId} suspended` }),
    };
  } catch (error) {
    console. error('Suspend merchant error:', error);
    return {
      statusCode: 500,
      body: JSON.stringify({ error: 'Failed to suspend merchant' }),
    };
  }
};

// ============================================================================
// DISPUTE RESOLUTION
// ============================================================================

export const resolveDispute: APIGatewayProxyHandlerV2 = async (event) => {
  if (!validateAdminAccess(event)) {
    return { statusCode: 403, body: JSON.stringify({ error: 'Access denied' }) };
  }

  try {
    const disputeId = event.pathParameters?. disputeId;
    const body = JSON.parse(event.body || '{}');
    const { resolution, amount, notes } = body;

    await ddb.send(
      new UpdateItemCommand({
        TableName: 'Disputes',
        Key: { PK: { S: disputeId! }, SK: { S: 'METADATA' } },
        UpdateExpression: 'SET #status = :status, resolution = :resolution, refundAmount = :amount, notes = :notes, resolvedAt = :now',
        ExpressionAttributeNames: { '#status': 'status' },
        ExpressionAttributeValues: {
          ':status': { S: 'RESOLVED' },
          ':resolution': { S: resolution },
          ':amount': { N: String(amount) },
          ':notes': { S: notes },
          ':now': { S: new Date().toISOString() },
        },
      }),
    );

    return {
      statusCode: 200,
      body: JSON.stringify({ message: `Dispute ${disputeId} resolved` }),
    };
  } catch (error) {
    console. error('Resolve dispute error:', error);
    return {
      statusCode: 500,
      body: JSON.stringify({ error: 'Failed to resolve dispute' }),
    };
  }
};
