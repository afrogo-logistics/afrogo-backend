import { APIGatewayProxyHandlerV2 } from 'aws-lambda';
import { DynamoDBClient, PutItemCommand, UpdateItemCommand, QueryCommand } from '@aws-sdk/client-dynamodb';
import { SQSClient, SendMessageCommand } from '@aws-sdk/client-sqs';
import { v4 as uuidv4 } from 'uuid';

const ddb = new DynamoDBClient({});
const sqs = new SQSClient({});

// ============================================================================
// CREATE SUPPORT TICKET
// ============================================================================

export const createTicket: APIGatewayProxyHandlerV2 = async (event) => {
  try {
    const body = JSON.parse(event.body || '{}');
    const { userId, userType, category, title, description, attachments } = body;

    if (!userId || !category || !title) {
      return {
        statusCode: 400,
        body: JSON.stringify({ error: 'Missing required fields' }),
      };
    }

    const ticketId = `TICKET#${uuidv4()}`;
    const now = new Date(). toISOString();

    await ddb.send(
      new PutItemCommand({
        TableName: 'SupportTickets',
        Item: {
          PK: { S: ticketId },
          SK: { S: 'METADATA' },
          ticketId: { S: ticketId },
          userId: { S: userId },
          userType: { S: userType },
          category: { S: category },
          title: { S: title },
          description: { S: description },
          attachments: { S: JSON.stringify(attachments || []) },
          status: { S: 'OPEN' },
          priority: { S: 'NORMAL' },
          createdAt: { S: now },
          updatedAt: { S: now },
          responses: { N: '0' },
        },
      }),
    );

    // Queue for agent assignment
    await sqs.send(
      new SendMessageCommand({
        QueueUrl: process.env.SUPPORT_ASSIGNMENT_QUEUE_URL || '',
        MessageBody: JSON.stringify({
          action: 'ASSIGN_TICKET',
          ticketId,
          category,
          priority: 'NORMAL',
        }),
      }),
    );

    return {
      statusCode: 201,
      body: JSON.stringify({
        ticketId,
        message: 'Support ticket created.  We will respond shortly.',
      }),
    };
  } catch (error) {
    console.error('Create ticket error:', error);
    return {
      statusCode: 500,
      body: JSON.stringify({ error: 'Failed to create ticket' }),
    };
  }
};

// ============================================================================
// GET TICKET DETAILS
// ============================================================================

export const getTicket: APIGatewayProxyHandlerV2 = async (event) => {
  try {
    const ticketId = event.pathParameters?.ticketId;

    const res = await ddb.send(
      new QueryCommand({
        TableName: 'SupportTickets',
        KeyConditionExpression: 'PK = :pk',
        ExpressionAttributeValues: {
          ':pk': { S: ticketId!  },
        },
      }),
    );

    if (! res.Items || res.Items. length === 0) {
      return {
        statusCode: 404,
        body: JSON. stringify({ error: 'Ticket not found' }),
      };
    }

    const ticket = res.Items[0];
    const responses = await ddb.send(
      new QueryCommand({
        TableName: 'SupportTickets',
        KeyConditionExpression: 'PK = :pk AND begins_with(SK, :sk)',
        ExpressionAttributeValues: {
          ':pk': { S: ticketId! },
          ':sk': { S: 'RESPONSE#' },
        },
      }),
    );

    return {
      statusCode: 200,
      body: JSON.stringify({
        ticket: {
          ticketId: ticket. ticketId. S,
          title: ticket.title.S,
          description: ticket.description.S,
          status: ticket.status.S,
          priority: ticket.priority.S,
          createdAt: ticket.createdAt.S,
          responses: responses.Items?. length || 0,
        },
        conversationHistory: responses.Items?.map(r => ({
          author: r.author.S,
          message: r.message.S,
          createdAt: r.createdAt.S,
        })) || [],
      }),
    };
  } catch (error) {
    console.error('Get ticket error:', error);
    return {
      statusCode: 500,
      body: JSON.stringify({ error: 'Failed to retrieve ticket' }),
    };
  }
};

// ============================================================================
// ADD TICKET RESPONSE
// ============================================================================

export const addTicketResponse: APIGatewayProxyHandlerV2 = async (event) => {
  try {
    const ticketId = event.pathParameters?. ticketId;
    const body = JSON.parse(event.body || '{}');
    const { authorId, authorType, message } = body;

    if (!ticketId || !authorId || ! message) {
      return {
        statusCode: 400,
        body: JSON.stringify({ error: 'Missing required fields' }),
      };
    }

    const responseId = `RESPONSE#${uuidv4()}`;
    const now = new Date().toISOString();

    await ddb.send(
      new PutItemCommand({
        TableName: 'SupportTickets',
        Item: {
          PK: { S: ticketId },
          SK: { S: responseId },
          responseId: { S: responseId },
          author: { S: authorId },
          authorType: { S: authorType },
          message: { S: message },
          createdAt: { S: now },
        },
      }),
    );

    // Update ticket
    await ddb.send(
      new UpdateItemCommand({
        TableName: 'SupportTickets',
        Key: { PK: { S: ticketId }, SK: { S: 'METADATA' } },
        UpdateExpression: 'SET #responses = #responses + :inc, #updated = :now',
        ExpressionAttributeNames: {
          '#responses': 'responses',
          '#updated': 'updatedAt',
        },
        ExpressionAttributeValues: {
          ':inc': { N: '1' },
          ':now': { S: now },
        },
      }),
    );

    return {
      statusCode: 201,
      body: JSON. stringify({ message: 'Response added' }),
    };
  } catch (error) {
    console. error('Add response error:', error);
    return {
      statusCode: 500,
      body: JSON.stringify({ error: 'Failed to add response' }),
    };
  }
};

// ============================================================================
// SUBMIT FEEDBACK
// ============================================================================

export const submitFeedback: APIGatewayProxyHandlerV2 = async (event) => {
  try {
    const body = JSON.parse(event.body || '{}');
    const { userId, rating, category, message } = body;

    if (!userId || !rating) {
      return {
        statusCode: 400,
        body: JSON.stringify({ error: 'User ID and rating required' }),
      };
    }

    const feedbackId = `FEEDBACK#${uuidv4()}`;
    const now = new Date().toISOString();

    await ddb.send(
      new PutItemCommand({
        TableName: 'Feedback',
        Item: {
          PK: { S: userId },
          SK: { S: feedbackId },
          feedbackId: { S: feedbackId },
          rating: { N: String(rating) },
          category: { S: category || 'GENERAL' },
          message: { S: message || '' },
          createdAt: { S: now },
        },
      }),
    );

    return {
      statusCode: 201,
      body: JSON.stringify({ message: 'Feedback submitted.  Thank you!' }),
    };
  } catch (error) {
    console.error('Submit feedback error:', error);
    return {
      statusCode: 500,
      body: JSON.stringify({ error: 'Failed to submit feedback' }),
    };
  }
};
