// SHOPIFY WEBHOOK HANDLER
// ============================================================================

export const shopifyWebhookHandler: APIGatewayProxyHandlerV2 = async (event) => {
  try {
    const hmacHeader = event.headers['x-shopify-hmac-sha256'];
    const shopifyStoreId = event.headers['x-shopify-shop-api-access-token'];
    const body = event.body || '';

    // Verify webhook authenticity
    const hmac = crypto
      .createHmac('sha256', process.env.SHOPIFY_API_SECRET || '')
      .update(body, 'utf8')
      . digest('base64');

    if (hmac !== hmacHeader) {
      console.warn('Shopify webhook HMAC validation failed');
      return { statusCode: 403, body: JSON.stringify({ error: 'Invalid signature' }) };
    }

    const data = JSON.parse(body);
    const topic = event.headers['x-shopify-topic'];

    console.log(`Shopify webhook received: ${topic}`);

    // Route to appropriate handler
    if (topic === 'orders/create' || topic === 'orders/updated') {
      await handleShopifyOrder(data, shopifyStoreId);
    } else if (topic === 'fulfillments/update') {
      await handleShopifyFulfillment(data, shopifyStoreId);
    }

    return { statusCode: 200, body: JSON.stringify({ success: true }) };
  } catch (error) {
    console. error('Shopify webhook error:', error);
    return { statusCode: 500, body: JSON.stringify({ error: 'Webhook processing failed' }) };
  }
};

// ============================================================================
// HANDLE SHOPIFY ORDER
// ============================================================================

async function handleShopifyOrder(order: any, shopifyStoreId: string) {
  try {
    // Find linked merchant
    const merchantId = `MERCHANT#${shopifyStoreId}`; // Simplified mapping

    // Extract shipping address
    const shipping = order.shipping_address;
    if (!shipping) return;

    // Create parcels for each line item
    for (const item of order.line_items || []) {
      const parcelData = {
        merchantId,
        recipientName: `${shipping.first_name} ${shipping.last_name}`,
        recipientPhone: shipping.phone,
        recipientEmail: order.email,
        recipientAddress: `${shipping.address1}, ${shipping.city}`,
        recipientLat: 0, // Will be geocoded
        recipientLng: 0,
        weight: (item.grams || 0) / 1000,
        contents: item.title,
        serviceType: 'AFROGO_DOOR',
        orderId: order.order_number,
        reference: `SHOPIFY#${order.id}#${item.id}`,
        insurance: false,
      };

      // Queue for parcel creation
      await sqs.send(
        new SendMessageCommand({
          QueueUrl: process.env. PARCEL_CREATION_QUEUE_URL || '',
          MessageBody: JSON. stringify({
            action: 'CREATE_PARCEL',
            payload: parcelData,
          }),
        }),
      );
    }

    // Update Shopify order with tracking info
    await updateShopifyOrderWithTracking(shopifyStoreId, order. id, order. order_number);
  } catch (error) {
    console. error('Handle Shopify order error:', error);
  }
}

            {
              fulfillment_order_line_item_id: 0, // Would be set per item

          ],
          tracking_info: {
            number: `AFR${orderNumber}`,

            url: `${process.env.TRACKING_URL}/SHOPIFY#${orderId}`,

      },

    // Verify webhook (simplified - use WooCommerce SDK in production)

// ============================================================================
// CUSTOM API WEBHOOK HANDLER
// ============================================================================

export const customWebhookHandler: APIGatewayProxyHandlerV2 = async (event) => {
  try {
    const body = JSON.parse(event. body || '{}');
    const { merchantId, shipments } = body;

    if (!merchantId || !Array.isArray(shipments)) {
      return {
        statusCode: 400,
        body: JSON.stringify({ error: 'merchantId and shipments array required' }),
      };
    }

    // Process each shipment
    for (const shipment of shipments) {
      const parcelData = {
        merchantId,
        recipientName: shipment.recipientName,
        recipientPhone: shipment.recipientPhone,
        recipientAddress: shipment.recipientAddress,
        recipientLat: shipment.recipientLat,
        recipientLng: shipment.recipientLng,
        weight: shipment.weight || 1,
        contents: shipment.contents,
        serviceType: shipment.serviceType || 'AFROGO_DOOR',
        orderId: shipment.orderId,
        reference: shipment.reference,
        insurance: shipment.insurance || false,
      };

      await sqs. send(
        new SendMessageCommand({
          QueueUrl: process.env.PARCEL_CREATION_QUEUE_URL || '',
          MessageBody: JSON.stringify({
            action: 'CREATE_PARCEL',
            payload: parcelData,
          }),
        }),
      );
    }

    return {
      statusCode: 201,
      body: JSON.stringify({
        message: 'Shipments queued for processing',
        count: shipments.length,
      }),
    };
  } catch (error) {
    console. error('Custom webhook error:', error);
    return {
      statusCode: 500,
      body: JSON.stringify({ error: 'Webhook processing failed' }),
    };
  }
};
