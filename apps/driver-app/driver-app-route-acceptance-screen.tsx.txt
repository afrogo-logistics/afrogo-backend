/**
 * DRIVER APP - Route Acceptance Screen
 * 
 * React Native component displaying:
 *   - Zone & geometry (stops, distance, duration)
 *   - Fixed payout with breakdown
 *   - Model C transparency (base + distance component)
 *   - Accept/Decline buttons
 * 
 * Status: PRODUCTION
 * Framework: React Native + Redux
 */

import React, { useState } from 'react';
import {
  View,
  Text,
  TouchableOpacity,
  ScrollView,
  StyleSheet,
  ActivityIndicator,
  Alert,
} from 'react-native';
import { useDispatch, useSelector } from 'react-redux';

interface RouteOffer {
  routeId: string;
  zoneName: string;
  stopCount: number;
  plannedDistanceKm: number;
  estimatedDurationMin: number;
  financials: {
    totalEarnings: number;
    ratePerParcel: number;
    currency: string;
    modelVersion: string;
    baseRate: number;
    kmFactor: number;
    distanceComponent: number;
  };
  confidence: {
    isGuaranteed: boolean;
    label: string;
  };
}

interface Props {
  route: { params: { offer: RouteOffer } };
  navigation: any;
}

export const RouteAcceptanceScreen: React.FC<Props> = ({ route, navigation }) => {
  const { offer } = route.params;
  const dispatch = useDispatch();
  const { user } = useSelector((state: any) => state.auth);

  const [isAccepting, setIsAccepting] = useState(false);

  const handleAccept = async () => {
    if (!user?. driverId) {
      Alert. alert('Error', 'Driver ID not found');
      return;
    }

    setIsAccepting(true);

    try {
      const response = await fetch(
        `${process.env.API_URL}/driver/routes/${offer.routeId}/accept`,
        {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            Authorization: `Bearer ${user.authToken}`,
          },
          body: JSON.stringify({
            driverId: user.driverId,
            acceptedAt: new Date().toISOString(),
          }),
        },
      );

      if (!response.ok) {
        throw new Error('Failed to accept route');
      }

      // Navigate to active route
      navigation.replace('ActiveRoute', { routeId: offer.routeId });
    } catch (error) {
      console.error('Error accepting route:', error);
      Alert.alert('Error', 'Failed to accept route.  Please try again.');
    } finally {
      setIsAccepting(false);
    }
  };

  const hourlyRate = (offer.financials.totalEarnings / offer.estimatedDurationMin) * 60;

  return (
    <ScrollView style={styles.container} showsVerticalScrollIndicator={false}>
      {/* HEADER */}
      <View style={styles.header}>
        <Text style={styles.title}>Route Offer</Text>
        <TouchableOpacity onPress={() => navigation.goBack()}>
          <Text style={styles.closeBtn}>✕</Text>
        </TouchableOpacity>
      </View>

      {/* ROUTE INFO CARD */}
      <View style={styles.card}>
        <Text style={styles.sectionTitle}>{offer.zoneName}</Text>

        <View style={styles.infoRow}>
          <View style={styles.infoPair}>
            <Text style={styles.label}>Stops</Text>
            <Text style={styles.value}>{offer. stopCount}</Text>
          </View>
          <View style={styles.infoPair}>
            <Text style={styles.label}>Distance</Text>
            <Text style={styles.value}>{offer. plannedDistanceKm.toFixed(1)} km</Text>
          </View>
          <View style={styles.infoPair}>
            <Text style={styles.label}>Duration</Text>
            <Text style={styles.value}>
              {Math.floor(offer.estimatedDurationMin / 60)}h {offer.estimatedDurationMin % 60}m
            </Text>
          </View>
        </View>
      </View>

      {/* EARNINGS CARD - PROMINENT */}
      <View style={[styles.card, styles.earningsCard]}>
        <View style={styles.earningsHeader}>
          <Text style={styles.earningsLabel}>Fixed Earnings</Text>
          <View style={styles.badge}>
            <Text style={styles.badgeText}>LOCKED</Text>
          </View>
        </View>

        {/* TOTAL AMOUNT */}
        <View style={styles.amountContainer}>
          <Text style={styles.currency}>{offer.financials.currency}</Text>
          <Text style={styles.amount}>
            {offer.financials.totalEarnings.toFixed(2)}
          </Text>
        </View>

        {/* HOURLY EQUIVALENT */}
        <Text style={styles.hourlyRate}>
          ≈ {offer.financials.currency} {hourlyRate.toFixed(2)}/hour
        </Text>

        {/* MODEL C BREAKDOWN */}
        <View style={styles.breakdownSection}>
          <Text style={styles.breakdownTitle}>How we calculated this:</Text>

          <View style={styles.breakdownItem}>
            <Text style={styles.breakdownLabel}>Base rate (per stop)</Text>
            <Text style={styles.breakdownValue}>
              {offer.financials. currency} {offer.financials.baseRate.toFixed(2)} × {offer.stopCount}
            </Text>
            <Text style={styles.breakdownResult}>
              = {offer.financials.currency}{' '}
              {(offer.financials.baseRate * offer.stopCount).toFixed(2)}
            </Text>
          </View>

          <View style={styles.breakdownItem}>
            <Text style={styles.breakdownLabel}>Distance allowance</Text>
            <Text style={styles.breakdownValue}>
              {offer.financials. currency} {offer.financials. kmFactor.toFixed(3)}/km × {offer.plannedDistanceKm.toFixed(1)} km
            </Text>
            <Text style={styles.breakdownResult}>
              = {offer. financials.currency} {offer.financials.distanceComponent.toFixed(2)}
            </Text>
          </View>

          <View style={styles.divider} />

          <View style={styles.totalBreakdown}>
            <Text style={styles.breakdownLabel}>Rate per stop</Text>
            <Text style={styles.breakdownValue}>
              {offer.financials.currency} {offer.financials.ratePerParcel.toFixed(2)}
            </Text>
          </View>

          <View style={styles.totalBreakdown}>
            <Text style={styles.breakdownLabel}>Total ({offer.stopCount} stops)</Text>
            <Text style={styles.breakdownValueBold}>
              {offer.financials.currency} {offer.financials.totalEarnings.toFixed(2)}
            </Text>
          </View>
        </View>

        {/* GUARANTEE STATEMENT */}
        <View style={styles.guaranteeBox}>
          <Text style={styles.guaranteeIcon}>✓</Text>
          <Text style={styles.guaranteeText}>
            This amount is guaranteed.  You'll earn{' '}
            <Text style={styles.bold}>{offer.financials.currency} {offer.financials.totalEarnings.toFixed(2)}</Text> for
            completing all {offer.stopCount} stops, regardless of the actual route taken.
          </Text>
        </View>

        {/* CONFIDENCE LABEL */}
        <View style={styles.confidenceBox}>
          <Text style={styles.confidenceText}>{offer.confidence.label}</Text>
        </View>
      </View>

      {/* TERMS LINK */}
      <TouchableOpacity style={styles.termsLink}>
        <Text style={styles. termsText}>View detailed route & terms</Text>
      </TouchableOpacity>

      {/* ACTION BUTTONS */}
      <View style={styles.actionContainer}>
        <TouchableOpacity
          style={[styles.button, styles.declineButton]}
          onPress={() => navigation.goBack()}
          disabled={isAccepting}
        >
          <Text style={styles.buttonText}>Decline</Text>
        </TouchableOpacity>

        <TouchableOpacity
          style={[styles. button, styles.acceptButton]}
          onPress={handleAccept}
          disabled={isAccepting}
        >
          {isAccepting ? (
            <ActivityIndicator color="#fff" />
          ) : (
            <Text style={styles.buttonText}>Accept Route</Text>
          )}
        </TouchableOpacity>
      </View>

      <View style={styles.spacer} />
    </ScrollView>
  );
};
